PYTHON
"""64-cell reference kernel. Exact integers only. No floats."""
from concurrent.futures import ThreadPoolExecutor

N = 64
SIDE = 8

def index(row, col):
    return row * SIDE + col

def place(i):
    return divmod(i, SIDE)

def is_boundary(i):
    row, col = place(i)
    return row == 0 or row == SIDE - 1 or col == 0 or col == SIDE - 1

def cell_value(i):
    return i * 17 - 40

def forward(i, value):
    mark = i if is_boundary(i) else -i
    return value + mark

def inverse(i, value):
    mark = i if is_boundary(i) else -i
    return value - mark

def serial_roundtrip(src):
    mid = [forward(i, src[i]) for i in range(N)]
    back = [inverse(i, mid[i]) for i in range(N)]
    return back

def parallel_roundtrip(src):
    def one(i):
        return inverse(i, forward(i, src[i]))
    out = [None] * N
    with ThreadPoolExecutor(max_workers=N) as pool:
        for i, value in enumerate(pool.map(one, range(N))):
            out[i] = value
    return out

def udn(need, src):
    if need != "roundtrip+parallel+readback":
        return None
    serial = serial_roundtrip(src)
    parallel = parallel_roundtrip(src)
    if serial != src or parallel != src or serial != parallel:
        return None
    return serial
